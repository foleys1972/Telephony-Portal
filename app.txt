def admin_required(f):
    @wraps(f)
    def decorated_function(*args, **kwargs):
        if not current_user.is_authenticated or not current_user.is_admin():
            flash('Admin access required.', 'danger')
            return redirect(url_for('index'))
        return f(*args, **kwargs)
    return decorated_function

def search_all_string_columns(model, search_term):
    """Create a filter that searches across all string columns of a model."""
    search_conditions = []
    for column in model.__table__.columns:
        if isinstance(column.type, db.String):
            search_conditions.append(column.like(f'%{search_term}%'))
    return db.or_(*search_conditions)

def suggest_field_mappings(csv_headers, expected_fields):
    """Suggest field mappings based on header similarity"""
    mappings = {}
    
    # Clean headers for comparison
    clean_headers = {header.lower().replace(' ', '_').replace('-', '_'): header for header in csv_headers}
    
    for field in expected_fields:
        # Direct match
        if field in clean_headers:
            mappings[field] = clean_headers[field]
            continue
        
        # Partial matches
        field_lower = field.lower()
        for clean_header, original_header in clean_headers.items():
            if field_lower in clean_header or clean_header in field_lower:
                mappings[field] = original_header
                break
        
        # Special cases
        if field == 'mac_address' and 'name' in clean_headers:
            mappings[field] = clean_headers['name']
        elif field == 'ip_address' and 'ipaddress' in clean_headers:
            mappings[field] = clean_headers['ipaddress']
        elif field == 'firmware_version' and 'firmwareversion' in clean_headers:
            mappings[field] = clean_headers['firmwareversion']
    
    return mappings

def process_import_data(entity_type, file_path, encoding, field_mappings):
    """Process import data for any entity type"""
    results = {'added': 0, 'updated': 0, 'skipped': 0}
    
    with open(file_path, 'r', encoding=encoding) as f:
        reader = csv.DictReader(f)
        
        for row in reader:
            try:
                if entity_type == 'ddi':
                    results = process_ddi_row(row, field_mappings, results)
                elif entity_type == 'private_wire':
                    results = process_pw_row(row, field_mappings, results)
                elif entity_type == 'turret':
                    results = process_turret_row(row, field_mappings, results)
            except Exception as e:
                print(f"Error processing row: {e}")
                results['skipped'] += 1
                continue
    
    db.session.commit()
    return results

def process_ddi_row(row, field_mappings, results):
    """Process a single DDI row - FIXED to always process records"""
    # Get DDI number
    ddi_field = field_mappings.get('ddi_number')
    if not ddi_field or ddi_field not in row:
        results['skipped'] += 1
        return results
    
    ddi_number = row[ddi_field].strip()
    
    # Validate DDI format
    if not (ddi_number.startswith('6') and len(ddi_number) == 12 and ddi_number.isdigit()):
        results['skipped'] += 1
        return results
    
    # Check if exists
    existing = DDINumber.query.filter_by(ddi_number=ddi_number).first()
    
    # Create data dictionary
    data = {}
    for db_field, csv_field in field_mappings.items():
        if csv_field in row and row[csv_field].strip():
            data[db_field] = row[csv_field].strip()
    
    if existing:
        # FIXED: Always update existing records, don't skip
        updated_fields = []
        for field, value in data.items():
            if field != 'ddi_number':
                old_value = getattr(existing, field)
                if str(old_value) != str(value):  # Only update if different
                    setattr(existing, field, value)
                    updated_fields.append(field)
        
        if updated_fields:
            existing.last_updated = datetime.utcnow()
            existing.last_change = f'Updated from import by {current_user.username} - fields: {", ".join(updated_fields)}'
            results['updated'] += 1
        else:
            # Even if no changes, count as processed (not skipped)
            results['updated'] += 1
    else:
        # Create new record
        new_ddi = DDINumber(
            last_change=f'Added from import by {current_user.username}',
            **data
        )
        if 'status' not in data:
            new_ddi.status = 'Active'
        db.session.add(new_ddi)
        results['added'] += 1
    
    return results

def save_import_template(entity_type, template_name, field_mappings):
    """Save import template for future use"""
    # Check if template name already exists
    existing = ImportTemplate.query.filter_by(
        name=template_name, 
        entity_type=entity_type
    ).first()
    
    if existing:
        existing.set_mappings(field_mappings)
        existing.last_used = datetime.utcnow()
    else:
        template = ImportTemplate(
            name=template_name,
            entity_type=entity_type,
            created_by=current_user.username
        )
        template.set_mappings(field_mappings)
        db.session.add(template)

@app.context_processor
def inject_template_vars():
    """Inject common variables into all templates"""
    try:
        if current_user.is_authenticated:
            has_new_features_flag = has_new_features()
            pending_moves = 0
            
            if has_new_features_flag:
                try:
                    result = db.engine.execute("SELECT COUNT(*) FROM turret_move WHERE status='Planned'")
                    pending_moves = result.fetchone()[0]
                except:
                    pending_moves = 0
            
            return {
                'has_new_features': has_new_features_flag,
                'pending_moves': pending_moves
            }
    except:
        pass
    return {'has_new_features': False, 'pending_moves': 0}

@app.route('/login', methods=['GET', 'POST'])
def login():
    if request.method == 'POST':
        username = request.form.get('username')
        password = request.form.get('password')
        
        user = User.query.filter_by(username=username).first()
        
        if user and check_password_hash(user.password, password):
            if not user.active:
                flash('This account has been disabled. Please contact an administrator.', 'danger')
                return render_template('login.html')
                
            login_user(user)
            user.update_last_login()
            flash('Login successful!', 'success')
            return redirect(url_for('index'))
        else:
            flash('Invalid username or password', 'danger')
    
    return render_template('login.html')

@app.route('/admin/users/add', methods=['GET', 'POST'])
@login_required
@admin_required
def add_user():
    if request.method == 'POST':
        username = request.form.get('username')
        password = request.form.get('password')
        role = request.form.get('role')
        
        if User.query.filter_by(username=username).first():
            flash('Username already exists', 'danger')
            return redirect(url_for('add_user'))
        
        hashed_password = generate_password_hash(password)
        new_user = User(username=username, password=hashed_password, role=role)
        
        db.session.add(new_user)
        db.session.commit()
        
        flash('User added successfully', 'success')
        return redirect(url_for('manage_users'))
    
    return render_template('admin/add_user.html')

    return redirect(url_for('manage_users'))

        old_status = ddi.status
        
        ddi.username = request.form.get('username')
        ddi.line_type = request.form.get('line_type')
        ddi.cisco_cluster = request.form.get('cisco_cluster')
        ddi.tpo_dns_name = request.form.get('tpo_dns_name')

        ddi.status = request.form.get('status')

            # Parse the date
            planned_date = datetime.strptime(planned_date, '%Y-%m-%dT%H:%M')
            
            moves_created = 0
            
            # Create individual moves for each turret
            for move_data in moves_data:
                turret = DealerboardTurret.query.get(move_data['turret_id'])
                if not turret:
                    continue
                
                current_desk = turret.desk_location or 'Unknown'
                
                # Create the move record
                new_move = TurretMove(
                    turret_id=move_data['turret_id'],
                    current_desk=current_desk,
                    new_desk=move_data['new_desk'],
                    planned_date=planned_date,
                    notes=notes,
                    created_by=current_user.username,
                    status='Planned'
                )
                
                db.session.add(new_move)
                moves_created += 1
            
            db.session.commit()
            
            if moves_created == 1:
                flash(f'Move planned successfully for 1 turret!', 'success')
            else:
                flash(f'Moves planned successfully for {moves_created} turrets!', 'success')

            flash('Invalid date format.', 'danger')

    office_filter = request.args.get('office', '')
    search = request.args.get('search', '')
    
    query = DealerboardTurret.query
    
    if location_filter:
        query = query.filter(DealerboardTurret.country == location_filter)
    
    if office_filter:
        query = query.filter(DealerboardTurret.office.like(f'%{office_filter}%'))
    
    if search:

@app.route('/import/upload/<entity_type>', methods=['POST'])
@login_required
def upload_import_file(entity_type):
    """Upload and analyze CSV file for field mapping"""
    if 'file' not in request.files:
        flash('No file selected', 'danger')
        return redirect(url_for('dynamic_import', entity_type=entity_type))
    
    file = request.files['file']
    if file.filename == '':
        flash('No file selected', 'danger')
        return redirect(url_for('dynamic_import', entity_type=entity_type))
    
    try:
        # Save file temporarily
        filename = secure_filename(file.filename)
        temp_path = os.path.join(app.config['UPLOAD_FOLDER'], f"temp_{filename}")
        file.save(temp_path)
        
        # Try to read CSV with different encodings
        encodings_to_try = ['utf-8', 'latin-1', 'iso-8859-1', 'cp1252', 'windows-1252']
        csv_headers = None
        sample_data = []
        used_encoding = None
        
        for encoding in encodings_to_try:
            try:
                with open(temp_path, 'r', encoding=encoding) as f:
                    reader = csv.reader(f)
                    csv_headers = next(reader)  # Get header row
                    
                    # Get first 3 rows as sample
                    for i, row in enumerate(reader):
                        if i < 3:
                            sample_data.append(row)
                        else:
                            break
                    
                    used_encoding = encoding
                    break
            except UnicodeDecodeError:
                continue
        
        if not csv_headers:
            os.remove(temp_path)
            flash('Could not decode file. Please check file encoding.', 'danger')
            return redirect(url_for('dynamic_import', entity_type=entity_type))
        
        # Get expected fields for the entity type
        expected_fields = get_expected_fields(entity_type)
        
        # Get suggested mappings (match headers to expected fields)
        suggested_mappings = suggest_field_mappings(csv_headers, expected_fields)
        
        # Get saved templates for pre-fill option
        templates = ImportTemplate.query.filter_by(entity_type=entity_type).order_by(ImportTemplate.last_used.desc()).all()
        
        return render_template('import/field_mapping.html',
                             entity_type=entity_type,
                             file_path=temp_path,
                             encoding=used_encoding,
                             csv_headers=csv_headers,
                             sample_data=sample_data,
                             expected_fields=expected_fields,
                             suggested_mappings=suggested_mappings,
                             templates=templates)
        
    except Exception as e:
        if 'temp_path' in locals() and os.path.exists(temp_path):
            os.remove(temp_path)
        flash(f'Error processing file: {str(e)}', 'danger')
        return redirect(url_for('dynamic_import', entity_type=entity_type))

@app.route('/import/process/<entity_type>', methods=['POST'])
@login_required
def process_dynamic_import(entity_type):
    """Process the import with user-defined field mappings"""
    file_path = request.form.get('file_path')
    encoding = request.form.get('encoding')
    save_template = request.form.get('save_template')
    template_name = request.form.get('template_name', '').strip()
    
    if not file_path or not os.path.exists(file_path):
        flash('Invalid file information', 'danger')
        return redirect(url_for('dynamic_import', entity_type=entity_type))
    
    # Get field mappings from form
    expected_fields = get_expected_fields(entity_type)
    field_mappings = {}
    
    for field in expected_fields:
        csv_field = request.form.get(f'map_{field}')
        if csv_field and csv_field != '':
            field_mappings[field] = csv_field
    
    # Validate required mappings
    required_fields = get_required_fields(entity_type)
    missing_required = [field for field in required_fields if field not in field_mappings]
    
    if missing_required:
        flash(f'Required field mappings missing: {", ".join(missing_required)}', 'danger')
        return redirect(request.url)
    
    try:
        # Process the import
        results = process_import_data(entity_type, file_path, encoding, field_mappings)
        
        # Save template if requested
        if save_template and template_name:
            save_import_template(entity_type, template_name, field_mappings)
        
        # Clean up temp file
        if os.path.exists(file_path):
            os.remove(file_path)
        
        flash(f'Import completed: {results["added"]} added, {results["updated"]} updated, {results["skipped"]} skipped', 'success')
        
        # Redirect to appropriate list page
        if entity_type == 'ddi':
            return redirect(url_for('ddi_list'))
        elif entity_type == 'private_wire':
            return redirect(url_for('pw_list'))
        elif entity_type == 'turret':
            return redirect(url_for('turret_list'))
        
    except Exception as e:
        if os.path.exists(file_path):
            os.remove(file_path)
        flash(f'Error processing import: {str(e)}', 'danger')
        return redirect(url_for('dynamic_import', entity_type=entity_type))

@app.route('/admin/database/<table_name>/edit/<int:record_id>', methods=['GET', 'POST'])
@login_required
@admin_required
def admin_edit_record(table_name, record_id):
    """Edit a record in a database table - FIXED VERSION"""
    if not has_new_features():
        flash('Database management requires migration.', 'warning')
        return redirect(url_for('index'))
    
    # Map table names to models
    table_models = {
        'users': User,
        'ddi_numbers': DDINumber,
        'private_wires': PrivateWire,
        'turrets': DealerboardTurret,
        'ceased_wires': CeasedPrivateWire,
        'turret_moves': TurretMove,
        'import_templates': ImportTemplate,
        'audit_logs': AuditLog
    }
    
    if table_name not in table_models:
        flash(f'Table "{table_name}" not found.', 'danger')
        return redirect(url_for('admin_database'))
    
    model = table_models[table_name]
    
    try:
        # Find the record
        record = model.query.get_or_404(record_id)
        
        # Get column information
        columns = []
        for column in model.__table__.columns:
            col_info = {
                'name': column.name,
                'type': str(column.type),
                'nullable': column.nullable,
                'primary_key': column.primary_key,
                'value': getattr(record, column.name)
            }
            columns.append(col_info)
        
        if request.method == 'POST':
            try:
                # Update the record with proper data type handling
                for column in model.__table__.columns:
                    if not column.primary_key:  # Don't update primary keys
                        field_value = request.form.get(column.name)
                        
                        # Skip auto-updated fields unless explicitly provided
                        if column.name in ['last_updated', 'date_created'] and not field_value:
                            continue
                        
                        # Handle different data types properly
                        if field_value is not None:
                            # Convert empty strings to None for nullable fields
                            if field_value.strip() == '' and column.nullable:
                                field_value = None
                            elif field_value.strip() == '' and not column.nullable:
                                # Skip empty required fields - keep existing value
                                continue
                            else:
                                # Handle different column types
                                if 'BOOLEAN' in str(column.type).upper():
                                    field_value = field_value.lower() in ['true', '1', 'yes', 'on', 'checked']
                                elif 'INTEGER' in str(column.type).upper():
                                    try:
                                        field_value = int(field_value) if field_value.strip() else None
                                    except ValueError:
                                        flash(f'Invalid integer value for {column.name}', 'warning')
                                        continue
                                elif 'DATETIME' in str(column.type).upper():
                                    if field_value.strip():
                                        try:
                                            field_value = datetime.strptime(field_value, '%Y-%m-%dT%H:%M')
                                        except ValueError:
                                            try:
                                                field_value = datetime.strptime(field_value, '%Y-%m-%d %H:%M:%S')
                                            except ValueError:
                                                flash(f'Invalid date format for {column.name}', 'warning')
                                                continue
                                    else:
                                        field_value = None
                            
                            # Set the field value
                            setattr(record, column.name, field_value)
                
                # Update timestamp if available
                if hasattr(record, 'last_updated'):
                    record.last_updated = datetime.utcnow()
                
                # Commit the changes
                db.session.commit()
                flash(f'Record {record_id} updated successfully.', 'success')
                return redirect(url_for('admin_table_view', table_name=table_name))
                
            except Exception as e:
                db.session.rollback()
                flash(f'Error updating record: {str(e)}', 'danger')
        
        return render_template('admin/edit_record.html', 
                             table_name=table_name,
                             record=record,
                             record_id=record_id,
                             columns=columns)
        
    except Exception as e:
        flash(f'Error editing record: {str(e)}', 'danger')
        return redirect(url_for('admin_table_view', table_name=table_name))
